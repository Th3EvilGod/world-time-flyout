cmake_minimum_required(VERSION 3.24)
project(WorldTimeFlyout VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(WorldTimeFlyout WIN32 src/main.cpp)
target_compile_definitions(WorldTimeFlyout PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(WorldTimeFlyout PRIVATE /W4 /EHsc /utf-8 /O2)
  set_property(TARGET WorldTimeFlyout PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
target_link_libraries(WorldTimeFlyout PRIVATE user32 gdi32 gdiplus dwmapi shell32 advapi32 comctl32 ole32)
